# Milestone 1 (Data Foundation): synthetic dataset generation/validation.
# pandas/numpy are part of the documented tech stack (docs/11 S10) for
# exactly this purpose; Pillow renders the fictional/generic sample cheque
# and signature images consumed by later milestones.
pandas==3.0.5
# NOTE(vas): numpy 2.5.x is capped out by paddlex (a transitive dependency of
# the pinned paddleocr 3.7.0) which requires numpy<2.4, while opencv-python-
# headless requires numpy>=2. The installable overlap is [2.0, 2.4); relax the
# pin to that range so the full stack resolves on Python 3.13. PaddleOCR and
# Tesseract pins are left untouched.
numpy>=2.0,<2.4
Pillow==12.3.0

# Milestone 2 (Cheque Upload & Image Processing): OpenCV is the documented
# computer-vision library (docs/13 S18); opencv-python-headless is used
# instead of opencv-python since this runs in a server/backend context
# with no GUI. PyMuPDF renders the first page of an uploaded PDF cheque
# to an image (docs/12 S8) without requiring an external Poppler binary.
# PaddleX OCR-core installs opencv-contrib-python==4.10.0.84, which shares
# the cv2 import namespace. Keep both distributions on the same OpenCV
# version so whichever wheel supplies cv2 has consistent behavior.
opencv-python-headless==4.10.0.84
pymupdf==1.28.2

# OCR engines: PaddleOCR is the primary engine (CPU inference); PyTesseract
# and native Tesseract remain the automatic fallback. PaddleOCR downloads
# its English detection and recognition models on first use.
paddlepaddle==3.3.1
paddleocr==3.7.0

# PyTesseract is the Python wrapper for the native Tesseract OCR fallback.
# The Tesseract binary
# itself (v5.4.0, installed via `winget install UB-Mannheim.TesseractOCR`)
# is a separate, non-pip system dependency -- see app/core/config.py's
# tesseract_cmd_path setting and the Milestone 3 report for details.
pytesseract==0.3.13
